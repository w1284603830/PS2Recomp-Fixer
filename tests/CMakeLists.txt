cmake_minimum_required(VERSION 3.24)
project(FixerBoundaryTests LANGUAGES CXX)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(RT "${ROOT}/third_party/src/PS2Recomp/ps2xRuntime")
set(IOP "${ROOT}/third_party/src/PS2Recomp/ps2xIOP")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_boundary_fixtures.py" "${GENERATED}"
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(vfs_boundary vfs_boundary_tests.cpp "${RT}/src/lib/ps2_vfs.cpp" "${RT}/src/lib/ps2_rom_device.cpp" "${IOP}/src/ps2_path.cpp")
add_executable(memory_boundary memory_boundary_tests.cpp)
add_executable(vag_boundary vag_boundary_tests.cpp "${RT}/src/lib/ps2_audio.cpp" "${RT}/src/lib/ps2_audio_vag.cpp")
add_executable(overlay_registry overlay_registry_tests.cpp "${RT}/src/lib/ps2_overlays.cpp")
find_package(Threads REQUIRED)
target_link_libraries(overlay_registry PRIVATE Threads::Threads)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_encoding_fixtures.py" "${GENERATED}"
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(vfs_encoding vfs_encoding_tests.cpp "${RT}/src/lib/ps2_vfs.cpp" "${RT}/src/lib/ps2_rom_device.cpp" "${IOP}/src/ps2_path.cpp")
target_include_directories(vfs_encoding PRIVATE "${ROOT}/examples/generic_runner")
if(NOT WIN32)
    find_package(Iconv REQUIRED)
    target_link_libraries(vfs_encoding PRIVATE Iconv::Iconv)
endif()
foreach(target vfs_boundary memory_boundary vag_boundary vfs_encoding overlay_registry)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${RT}/include" "${IOP}/include" "${GENERATED}" "${CMAKE_CURRENT_SOURCE_DIR}/stubs/audio")
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /arch:AVX2)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
set(VFS_CASES create-preserve create-readonly missing permissions append truncate exclusive exclusive-race rom-overflow large-seek disc-readonly versions invalid mixed unicode)
foreach(case IN LISTS VFS_CASES)
    add_test(NAME "vfs-${case}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_vfs_fixture.py" "$<TARGET_FILE:vfs_boundary>" "${case}")
endforeach()
foreach(case segments spans vu gs mc-string sif-mirror sif-scratch-invalid sif-metadata)
    add_test(NAME "memory-${case}" COMMAND memory_boundary "${case}")
endforeach()
foreach(case header-bounds complete partial rate-empty transfer-overflow)
    add_test(NAME "vag-${case}" COMMAND vag_boundary "${case}")
endforeach()
foreach(case empty dispatch reject clear copy threads)
    add_test(NAME "overlay-${case}" COMMAND overlay_registry "${case}")
endforeach()
set_tests_properties(vfs-exclusive-race PROPERTIES TIMEOUT 20)
foreach(encoding utf8 sjis gbk big5)
    foreach(case read create save trail decode-invalid)
        add_test(NAME "encoding-${encoding}-${case}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_vfs_fixture.py" "$<TARGET_FILE:vfs_encoding>" "${case}" "${encoding}")
    endforeach()
endforeach()
foreach(encoding sjis gbk big5)
    add_test(NAME "encoding-${encoding}-raw-reject" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_vfs_fixture.py" "$<TARGET_FILE:vfs_encoding>" raw-reject "${encoding}")
endforeach()
foreach(encoding gbk big5)
    add_test(NAME "encoding-${encoding}-ambiguous" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_vfs_fixture.py" "$<TARGET_FILE:vfs_encoding>" ambiguous "${encoding}")
endforeach()
foreach(case utf8-invalid switches)
    add_test(NAME "encoding-${case}" COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_vfs_fixture.py" "$<TARGET_FILE:vfs_encoding>" "${case}" utf8)
endforeach()
if(WIN32)
    add_test(NAME encoding-windows-native COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run_vfs_fixture.py" "$<TARGET_FILE:vfs_encoding>" native utf8)
endif()
